#!/bin/bash
# ==============================================================================
# Script cài đặt môi trường và Huấn luyện SONG SONG 2 SEEDS cho M2FTrans
# Seeds: Seed 1234 & Seed 1098
# Model: M2FTrans2D (MyoPS380 — 300 Epochs — Không Validation)
# Repository: https://github.com/mintduckk24/M2FTrans.git
# Quản lý tiến trình: 2 session tmux riêng biệt (chạy ngầm song song, không lo đứt mạng)
# ==============================================================================

set -e

echo "=== [1/7] Kiểm tra GPU NVIDIA ==="
cd /content

if [ -d "/usr/lib64-nvidia" ]; then
    if ! grep -q "/usr/lib64-nvidia" /etc/ld.so.conf.d/nvidia.conf 2>/dev/null; then
        echo "/usr/lib64-nvidia" >> /etc/ld.so.conf.d/nvidia.conf 2>/dev/null || true
        ldconfig 2>/dev/null || true
    fi
    export LD_LIBRARY_PATH=/usr/lib64-nvidia:${LD_LIBRARY_PATH}
fi
if ! grep -q "/usr/lib64-nvidia" ~/.bashrc 2>/dev/null; then
    echo 'export LD_LIBRARY_PATH=/usr/lib64-nvidia:$LD_LIBRARY_PATH' >> ~/.bashrc 2>/dev/null || true
fi

if ! command -v nvidia-smi &> /dev/null; then
    echo "⚠️ CẢNH BÁO: Chưa tìm thấy NVIDIA GPU. Hãy kiểm tra lại Runtime > Change runtime type > GPU!"
else
    nvidia-smi --query-gpu=name,memory.total --format=csv,noheader || true
fi

# Chống phân mảnh bộ nhớ khi chạy song song 2 tiến trình trên cùng 1 GPU
export PYTORCH_CUDA_ALLOC_CONF="expandable_segments:True"

echo "=== [2/7] Mount Google Drive (Lưu Checkpoints & Logs an toàn) ==="
python3 -c "
import os
if not os.path.exists('/content/drive/MyDrive'):
    try:
        from google.colab import drive
        drive.mount('/content/drive')
        print('✅ Mount Google Drive thành công!')
    except Exception as e:
        print('⚠️ Không thể mount Drive tự động (bỏ qua nếu không dùng):', e)
" || true

echo "=== [3/7] Thiết lập Repository M2FTrans ==="
REPO_URL="https://github.com/mintduckk24/M2FTrans.git"

if [ -f "./train.py" ] && [ -d "./models" ]; then
    echo "✅ Đang đứng trực tiếp trong thư mục M2FTrans."
elif [ -d "/content/M2FTrans" ]; then
    echo "✅ Thư mục /content/M2FTrans đã tồn tại."
elif [ -f "/content/M2FTrans.zip" ]; then
    echo "✅ Phát hiện file M2FTrans.zip tại /content/. Đang giải nén..."
    unzip -q -o "/content/M2FTrans.zip" -d /content/
elif [ -f "/content/drive/MyDrive/M2FTrans.zip" ]; then
    echo "✅ Phát hiện file M2FTrans.zip trên Google Drive. Đang giải nén..."
    unzip -q -o "/content/drive/MyDrive/M2FTrans.zip" -d /content/
elif [ -d "/content/drive/MyDrive/M2FTrans" ]; then
    echo "✅ Phát hiện thư mục M2FTrans trên Google Drive. Đang sao chép vào /content/M2FTrans..."
    cp -r "/content/drive/MyDrive/M2FTrans" /content/M2FTrans
else
    echo "Đang clone repository từ $REPO_URL..."
    git clone "$REPO_URL" /content/M2FTrans || {
        echo "❌ Chưa thể clone từ $REPO_URL."
        exit 1
    }
fi

if [ -d "/content/M2FTrans" ]; then
    cd /content/M2FTrans
fi

echo "=== [4/7] Cài đặt Dependencies ==="
python3 -m pip install --upgrade pip -q
if [ -f "requirements.txt" ]; then
    python3 -m pip install -r requirements.txt gdown -q
else
    python3 -m pip install torch torchvision numpy scipy h5py PyYAML pandas nibabel SimpleITK medpy tqdm gdown einops timm -q
fi

echo "=== [5/7] Thiết lập Thư mục Checkpoint an toàn ==="
if [ -d "/content/drive/MyDrive" ]; then
    BASE_SAVE_DIR="/content/drive/MyDrive/M2FTrans/checkpoints"
else
    BASE_SAVE_DIR="/content/M2FTrans/checkpoints"
fi

SAVE_DIR_1234="$BASE_SAVE_DIR/seed1234"
SAVE_DIR_1098="$BASE_SAVE_DIR/seed1098"
mkdir -p "$SAVE_DIR_1234" "$SAVE_DIR_1098"

echo "=== [6/7] Tải Dataset MyoPS380 ==="
DATASET_FOLDER_ID="13T6892xy-pUwC1BSjGl4sRxoypMu-SIS"
DOWNLOAD_DIR="/content/dataset_download"
TARGET_DATASET_DIR="/content/M2FTrans/MyoPS380_dataset"
mkdir -p "$DOWNLOAD_DIR" "$TARGET_DATASET_DIR"

DRIVE_DATASET_DIR="/content/drive/MyDrive/MyoPS380_dataset"
if [ -d "$DRIVE_DATASET_DIR/Processed_data/bSSFP" ]; then
    LOCAL_DATASET="$DRIVE_DATASET_DIR/Processed_data"
elif [ -d "/content/drive/MyDrive/Processed_data/bSSFP" ]; then
    LOCAL_DATASET="/content/drive/MyDrive/Processed_data"
else
    if [ -z "$(ls -A "$DOWNLOAD_DIR" 2>/dev/null)" ] && [ -z "$(ls -A "$TARGET_DATASET_DIR" 2>/dev/null)" ]; then
        echo "Đang tải dataset từ Google Drive..."
        gdown --folder --id "$DATASET_FOLDER_ID" -O "$DOWNLOAD_DIR" --remaining-ok || true
    fi
    for zipfile in $(find "$DOWNLOAD_DIR" -maxdepth 2 -name "*.zip" 2>/dev/null); do
        command -v unzip &>/dev/null || (apt-get update -qq && apt-get install -y unzip)
        unzip -q -o "$zipfile" -d "$TARGET_DATASET_DIR/"
    done
    if [ -d "$DOWNLOAD_DIR/Processed_data" ]; then
        cp -r "$DOWNLOAD_DIR/Processed_data" "$TARGET_DATASET_DIR/"
    fi
    FOUND_BSSFP=$(find "$TARGET_DATASET_DIR" "$DOWNLOAD_DIR" -type d -name "bSSFP" 2>/dev/null | head -n 1)
    if [ -n "$FOUND_BSSFP" ]; then
        LOCAL_DATASET=$(dirname "$FOUND_BSSFP")
    else
        LOCAL_DATASET="/content/M2FTrans/MyoPS380_dataset/Processed_data"
    fi
fi
echo "✅ Đường dẫn dataset: $LOCAL_DATASET"

SPLIT_DIR="/content/M2FTrans/dataset/splits/myops380_no_val"

echo "=== [7/7] Cài đặt tmux và Khởi chạy Song song 2 Seeds ==="
command -v tmux &> /dev/null || (apt-get update -qq && apt-get install -y tmux)

# Dừng session cũ nếu có
tmux kill-session -t seed1234 2>/dev/null || true
tmux kill-session -t seed1098 2>/dev/null || true

# Khởi chạy Seed 1234 trong session tmux seed1234
tmux new-session -d -s seed1234 "cd /content/M2FTrans && export PYTORCH_CUDA_ALLOC_CONF='expandable_segments:True' && python3 train.py --data_dir '$LOCAL_DATASET' --split_dir '$SPLIT_DIR' --save_dir '$SAVE_DIR_1234' --seed 1234 --max_epochs 300 --batch_size 16 --accum_steps 1 --lr 0.0003333 --weight_decay 0.0001 --amp auto --resume 2>&1 | tee '$SAVE_DIR_1234/train.log'"

# Khởi chạy Seed 1098 trong session tmux seed1098
tmux new-session -d -s seed1098 "cd /content/M2FTrans && export PYTORCH_CUDA_ALLOC_CONF='expandable_segments:True' && python3 train.py --data_dir '$LOCAL_DATASET' --split_dir '$SPLIT_DIR' --save_dir '$SAVE_DIR_1098' --seed 1098 --max_epochs 300 --batch_size 16 --accum_steps 1 --lr 0.0003333 --weight_decay 0.0001 --amp auto --resume 2>&1 | tee '$SAVE_DIR_1098/train.log'"

echo ""
echo "🎉 ĐÃ KHỞI CHẠY THÀNH CÔNG 2 SEED SONG SONG:"
echo "   🟢 Seed 1234 đang chạy ngầm trong tmux session: 'seed1234' (Log: $SAVE_DIR_1234/train.log)"
echo "   🟢 Seed 1098 đang chạy ngầm trong tmux session: 'seed1098' (Log: $SAVE_DIR_1098/train.log)"
echo ""
echo "📋 CÁC LỆNH HỮU ÍCH ĐỂ THEO DÕI TIẾN ĐỘ:"
echo "   1. Xem tiến trình Seed 1234 trực tiếp:   tmux attach -t seed1234"
echo "      (Để thoát màn hình xem mà không tắt train: Bấm Ctrl+B rồi bấm D)"
echo ""
echo "   2. Xem tiến trình Seed 1098 trực tiếp:   tmux attach -t seed1098"
echo "      (Để thoát màn hình xem mà không tắt train: Bấm Ctrl+B rồi bấm D)"
echo ""
echo "   3. Xem log tiến trình liên tục:"
echo "      tail -f $SAVE_DIR_1234/train.log"
echo "      tail -f $SAVE_DIR_1098/train.log"
echo ""
echo "   4. Kiểm tra GPU đang chạy cả 2 tiến trình:  nvidia-smi"
