#!/bin/bash
# ==============================================================================
# Script cài đặt môi trường và huấn luyện cho M2FTrans trên Google Colab CLI
# Model: M2FTrans2D (Multi-Scale Multi-Modal Fusion Transformer)
# Repository: https://github.com/mintduckk24/M2FTrans.git (hoặc sao chép từ Drive)
# Benchmark: MyoPS380 (Protocol: myops380_train304_test76_last_v1)
# ==============================================================================

set -e

echo "=== [1/7] Kiểm tra và Kích hoạt GPU NVIDIA ==="
cd /content

# Tự động liên kết thư viện driver NVIDIA nếu có trên Colab (/usr/lib64-nvidia)
if [ -d "/usr/lib64-nvidia" ]; then
    if ! grep -q "/usr/lib64-nvidia" /etc/ld.so.conf.d/nvidia.conf 2>/dev/null; then
        echo "/usr/lib64-nvidia" >> /etc/ld.so.conf.d/nvidia.conf 2>/dev/null || true
        ldconfig 2>/dev/null || true
    fi
    export LD_LIBRARY_PATH=/usr/lib64-nvidia:${LD_LIBRARY_PATH}
fi
if ! grep -q "/usr/lib64-nvidia" ~/.bashrc 2>/dev/null; then
    echo 'export LD_LIBRARY_PATH=/usr/lib64-nvidia:$LD_LIBRARY_PATH' >> ~/.bashrc 2>/dev/null || true
fi

if ! command -v nvidia-smi &> /dev/null; then
    echo "⚠️ CẢNH BÁO: Chưa tìm thấy NVIDIA GPU. Hãy kiểm tra lại Runtime > Change runtime type > GPU trên Colab!"
else
    nvidia-smi --query-gpu=name,memory.total --format=csv,noheader || true
fi

echo "=== [2/7] Mount Google Drive (Lưu Checkpoint an toàn) ==="
python3 -c "
import os
if not os.path.exists('/content/drive/MyDrive'):
    try:
        from google.colab import drive
        drive.mount('/content/drive')
        print('✅ Mount Google Drive thành công!')
    except Exception as e:
        print('⚠️ Không thể mount Drive tự động (bỏ qua nếu không dùng):', e)
" || true

echo "=== [3/7] Thiết lập Repository M2FTrans ==="
REPO_URL="https://github.com/mintduckk24/M2FTrans.git"

if [ -f "./train.py" ] && [ -d "./models" ]; then
    echo "✅ Đang đứng trực tiếp trong thư mục M2FTrans."
elif [ -d "/content/M2FTrans" ]; then
    echo "✅ Thư mục /content/M2FTrans đã tồn tại."
elif [ -f "/content/M2FTrans.zip" ]; then
    echo "✅ Phát hiện file M2FTrans.zip tại /content/. Đang giải nén..."
    unzip -q -o "/content/M2FTrans.zip" -d /content/
elif [ -f "/content/drive/MyDrive/M2FTrans.zip" ]; then
    echo "✅ Phát hiện file M2FTrans.zip trên Google Drive. Đang giải nén..."
    unzip -q -o "/content/drive/MyDrive/M2FTrans.zip" -d /content/
elif [ -d "/content/drive/MyDrive/M2FTrans" ]; then
    echo "✅ Phát hiện thư mục M2FTrans trên Google Drive. Đang sao chép vào /content/M2FTrans..."
    cp -r "/content/drive/MyDrive/M2FTrans" /content/M2FTrans
else
    echo "Đang clone repository từ $REPO_URL..."
    git clone "$REPO_URL" /content/M2FTrans || {
        echo "❌ Chưa thể clone từ $REPO_URL."
        echo "👉 Bạn có thể kéo thả file M2FTrans.zip vào khung Explorer của VS Code (/content/) rồi chạy lại."
        exit 1
    }
fi

if [ -d "/content/M2FTrans" ]; then
    cd /content/M2FTrans
fi

echo "=== [4/7] Cài đặt Dependencies cho M2FTrans ==="
python3 -m pip install --upgrade pip -q
if [ -f "requirements.txt" ]; then
    python3 -m pip install -r requirements.txt gdown -q
else
    python3 -m pip install torch torchvision numpy scipy h5py PyYAML pandas nibabel SimpleITK medpy tqdm gdown einops timm -q
fi

echo "=== [5/7] Thiết lập Thư mục Checkpoint an toàn trên Google Drive ==="
# M2FTrans khởi tạo trọng số ngẫu nhiên Kaiming / trunc_normal, không dùng pretrained ViT (train from scratch theo SCAR benchmark).
# Thiết lập thư mục lưu checkpoint trên Google Drive để tránh mất tiến trình khi Colab ngắt kết nối.
DRIVE_SAVE_DIR="/content/drive/MyDrive/M2FTrans/checkpoints"
LOCAL_SAVE_DIR="/content/M2FTrans/checkpoints"

mkdir -p "$LOCAL_SAVE_DIR"
if [ -d "/content/drive/MyDrive" ]; then
    mkdir -p "$DRIVE_SAVE_DIR"
    echo "✅ Google Drive đã sẵn sàng. Checkpoint sẽ được lưu trực tiếp tại: $DRIVE_SAVE_DIR"
    SAVE_DIR="$DRIVE_SAVE_DIR"
else
    echo "⚠️ Chưa mount Google Drive, checkpoint sẽ lưu tạm tại: $LOCAL_SAVE_DIR"
    SAVE_DIR="$LOCAL_SAVE_DIR"
fi

echo "=== [6/7] Tải Dataset từ Thư mục Google Drive ==="
# Thư mục Google Drive chứa dữ liệu MyoPS380
DATASET_FOLDER_ID="13T6892xy-pUwC1BSjGl4sRxoypMu-SIS"
DOWNLOAD_DIR="/content/dataset_download"
TARGET_DATASET_DIR="/content/M2FTrans/MyoPS380_dataset"

mkdir -p "$DOWNLOAD_DIR"
mkdir -p "$TARGET_DATASET_DIR"

# Kiểm tra nếu dataset đã có sẵn trên Google Drive đã mount
DRIVE_DATASET_DIR="/content/drive/MyDrive/MyoPS380_dataset"
if [ -d "$DRIVE_DATASET_DIR/Processed_data/bSSFP" ]; then
    echo "✅ Phát hiện dataset sẵn trên Google Drive tại $DRIVE_DATASET_DIR/Processed_data"
    LOCAL_DATASET="$DRIVE_DATASET_DIR/Processed_data"
elif [ -d "/content/drive/MyDrive/Processed_data/bSSFP" ]; then
    echo "✅ Phát hiện dataset sẵn trên Google Drive tại /content/drive/MyDrive/Processed_data"
    LOCAL_DATASET="/content/drive/MyDrive/Processed_data"
else
    # Tải toàn bộ thư mục Google Drive nếu chưa tải
    if [ -z "$(ls -A "$DOWNLOAD_DIR" 2>/dev/null)" ] && [ -z "$(ls -A "$TARGET_DATASET_DIR" 2>/dev/null)" ]; then
        echo "Đang tải toàn bộ thư mục dataset từ Google Drive (Folder ID: $DATASET_FOLDER_ID)..."
        gdown --folder --id "$DATASET_FOLDER_ID" -O "$DOWNLOAD_DIR" --remaining-ok || {
            echo "⚠️ Tải folder qua gdown gặp cảnh báo, tiếp tục kiểm tra các tệp đã tải..."
        }
    fi

    # Tự động giải nén bất kỳ tệp zip nào có trong thư mục tải về
    for zipfile in $(find "$DOWNLOAD_DIR" -maxdepth 2 -name "*.zip" 2>/dev/null); do
        echo "Đang giải nén $zipfile vào $TARGET_DATASET_DIR ..."
        command -v unzip &>/dev/null || (apt-get update -qq && apt-get install -y unzip)
        unzip -q -o "$zipfile" -d "$TARGET_DATASET_DIR/"
    done

    # Nếu trong thư mục tải về đã là thư mục dữ liệu giải nén sẵn
    if [ -d "$DOWNLOAD_DIR/Processed_data" ]; then
        cp -r "$DOWNLOAD_DIR/Processed_data" "$TARGET_DATASET_DIR/"
    fi

    # Cơ chế tự động định vị thư mục Processed_data chuẩn (chứa bSSFP/train_npz, tránh nhầm sang Raw_data)
    FOUND_TRAIN_NPZ=$(find "$TARGET_DATASET_DIR" "$DOWNLOAD_DIR" /content -type d -name "train_npz" 2>/dev/null | grep -i "bSSFP" | head -n 1)
    if [ -n "$FOUND_TRAIN_NPZ" ]; then
        LOCAL_DATASET=$(dirname $(dirname "$FOUND_TRAIN_NPZ"))
        echo "✅ Đã xác định thư mục Processed_data chuẩn tại: $LOCAL_DATASET"
    elif [ -d "$TARGET_DATASET_DIR/Processed_data/bSSFP" ]; then
        LOCAL_DATASET="$TARGET_DATASET_DIR/Processed_data"
    elif [ -d "$DOWNLOAD_DIR/Processed_data/bSSFP" ]; then
        LOCAL_DATASET="$DOWNLOAD_DIR/Processed_data"
    else
        LOCAL_DATASET="/content/M2FTrans/MyoPS380_dataset/Processed_data"
        echo "⚠️ Đặt đường dẫn mặc định: $LOCAL_DATASET"
    fi
fi

# Định vị thư mục split chuẩn (myops380_train304_test76_last_v1)
SPLIT_DIR="/content/M2FTrans/dataset/splits/myops380_no_val"
if [ ! -f "$SPLIT_DIR/train.txt" ] || [ ! -f "$SPLIT_DIR/test_vol.txt" ]; then
    echo "⚠️ Không tìm thấy split trong repo, đặt mặc định..."
fi

echo "=== [7/7] Cài đặt tmux và Kiểm tra PyTorch ==="
command -v tmux &> /dev/null || (apt-get update -qq && apt-get install -y tmux)
python3 -c "import torch; print(f'PyTorch: {torch.__version__} | CUDA: {torch.cuda.is_available()} | GPU: {torch.cuda.get_device_name(0) if torch.cuda.is_available() else \"None\"}')"

echo "=== HOÀN TẤT THIẾT LẬP MÔI TRƯỜNG CHO M2FTRANS ==="
echo ""
echo "💡 HƯỚNG DẪN CHẠY HUẤN LUYỆN (HỖ TRỢ CHẠY ĐƠN HOẶC SONG SONG 2 SEED):"
echo ""
echo "🔹 CÁCH 1: Chạy trực tiếp (Seed 1234 mặc định, Effective Batch Size = 16):"
echo "  cd /content/M2FTrans"
echo "  python3 train.py --data_dir \"$LOCAL_DATASET\" --split_dir \"$SPLIT_DIR\" --save_dir \"$SAVE_DIR\" --seed 1234 --max_epochs 300 --batch_size 16 --accum_steps 1 --lr 0.0003333 --weight_decay 0.0001 --amp auto --resume"
echo ""
echo "🔹 CÁCH 2: Chạy SONG SONG 2 SEED trong 2 session tmux (Độc lập, không xung đột):"
echo "  - Terminal / Session 1 (Seed 1234):"
echo "      tmux new -s seed1234"
echo "      cd /content/M2FTrans"
echo "      python3 train.py --data_dir \"$LOCAL_DATASET\" --split_dir \"$SPLIT_DIR\" --save_dir \"$SAVE_DIR/seed1234\" --seed 1234 --max_epochs 300 --batch_size 16 --accum_steps 1 --lr 0.0003333 --weight_decay 0.0001 --amp auto --resume"
echo "      (Nhấn Ctrl+B rồi nhấn D để thoát ra ngoài)"
echo ""
echo "  - Terminal / Session 2 (Seed 1098):"
echo "      tmux new -s seed1098"
echo "      cd /content/M2FTrans"
echo "      python3 train.py --data_dir \"$LOCAL_DATASET\" --split_dir \"$SPLIT_DIR\" --save_dir \"$SAVE_DIR/seed1098\" --seed 1098 --max_epochs 300 --batch_size 16 --accum_steps 1 --lr 0.0003333 --weight_decay 0.0001 --amp auto --resume"
echo "      (Nhấn Ctrl+B rồi nhấn D để thoát ra ngoài)"
echo ""
echo "  - Xem lại tiến trình bất cứ lúc nào:"
echo "      tmux attach -t seed1234    # Hoặc: tmux attach -t seed1098"
echo ""
echo "  - Đánh giá kiểm thử sau khi hoàn thành 300 epochs:"
echo "      python3 evaluate.py --checkpoint \"$SAVE_DIR/m2ftrans_seed1234.pt\" --data_dir \"$LOCAL_DATASET\" --split_dir \"$SPLIT_DIR\" --output_dir \"/content/M2FTrans/test_predictions_seed1234\" --batch_size 8 --amp auto"
echo "      python3 evaluate.py --checkpoint \"$SAVE_DIR/m2ftrans_seed1098.pt\" --data_dir \"$LOCAL_DATASET\" --split_dir \"$SPLIT_DIR\" --output_dir \"/content/M2FTrans/test_predictions_seed1098\" --batch_size 8 --amp auto"
echo ""

# === LỆNH CHẠY HUẤN LUYỆN TRỰC TIẾP ===
cd /content/M2FTrans
SEED=1234
python3 train.py \
    --data_dir "$LOCAL_DATASET" \
    --split_dir "$SPLIT_DIR" \
    --save_dir "$SAVE_DIR" \
    --max_epochs 300 \
    --batch_size 16 \
    --accum_steps 1 \
    --lr 0.0003333 \
    --weight_decay 0.0001 \
    --seed $SEED \
    --amp auto \
    --resume

echo "=== HUẤN LUYỆN HOÀN TẤT! TIẾN HÀNH ĐÁNH GIÁ 76 TEST VOLUMES ==="
CKPT_FILE="$SAVE_DIR/m2ftrans_seed${SEED}.pt"
if [ -f "$CKPT_FILE" ]; then
    python3 evaluate.py \
        --checkpoint "$CKPT_FILE" \
        --data_dir "$LOCAL_DATASET" \
        --split_dir "$SPLIT_DIR" \
        --output_dir "/content/M2FTrans/test_predictions" \
        --batch_size 8 \
        --amp auto
else
    echo "⚠️ Checkpoint $CKPT_FILE chưa tìm thấy."
fi
